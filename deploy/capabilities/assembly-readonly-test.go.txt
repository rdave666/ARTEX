package server
import (
 "context"
 "database/sql"
 "strings"
 "testing"
 "github.com/Autumn-27/artex/agent"
 "github.com/Autumn-27/artex/db"
 actool "github.com/Autumn-27/norma/tool"
)
func TestCapabilityReadOnly(t *testing.T) {
 c,err:=sql.Open("pgx","host=/var/run/postgresql port=5434 user=artex dbname=artex default_transaction_read_only=on")
 if err!=nil {t.Fatal(err)};defer c.Close()
 if err=c.Ping();err!=nil {t.Fatal(err)}
 pg:= &db.DB{DB:c}
 wireTools(pg,nil)
 for _,role:=range []string{"planner","mainagent","worker","pentest","retester","auto"} {
  found:=false
  for _,tool:=range agent.ToolResolve()(context.Background(),role,actool.DefaultTools()) {
   if tool.Name()!="Bash" {continue};found=true
   desc:=tool.Description()
   if !strings.Contains(desc,"kali-capabilities") {t.Fatalf("%s missing guide",role)}
   technical:=role!="planner" && role!="mainagent"
   if strings.Contains(desc,"- nmap:")!=technical {t.Fatalf("%s incorrect technical hints",role)}
   t.Logf("%s guide=true technical=%v",role,technical)
  }
  if !found {t.Fatalf("%s lacks Bash",role)}
 }
}
